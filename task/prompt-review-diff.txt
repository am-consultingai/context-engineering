+import graphene
+from django.core.exceptions import ValidationError

+from ....giftcard import events

+from ...core.scalars import PositiveDecimal
+from ...core.types import GiftCardError, NonNullList
+from ...plugins.dataloaders import get_plugin_manager_promise
+from ..types import GiftCard as GiftCardType
+
+
+class GiftCardBalanceTransferInput(graphene.InputObjectType):
+    source_id = graphene.ID(required=True, description="Source gift card ID.")
+    target_id = graphene.ID(required=True, description="Target gift card ID.")
+    amount = PositiveDecimal(required=True, description="Amount to move.")
+
+
+class GiftCardBalanceTransfer(BaseMutation):
+    gift_cards = NonNullList(GiftCardType, description="Updated gift cards.")
+
+    class Arguments:
+        transfers = NonNullList(
+            GiftCardBalanceTransferInput,
+            required=True,
+            description="Transfers to perform.",
+        )
+
+    class Meta:
+        description = "Transfers balance between gift cards."
+        doc_category = DOC_CATEGORY_GIFT_CARDS
+        error_type_class = GiftCardError
+        error_type_field = "gift_card_errors"
+
+    @classmethod
+    def perform_mutation(cls, _root, info: ResolveInfo, /, *, transfers):
+        user = info.context.user
+        if not user or not user.has_perm(GiftcardPermissions.MANAGE_GIFT_CARD):
+            raise ValidationError("Permission denied.")
+        manager = get_plugin_manager_promise(info.context).get()
+        updated = []
+        for transfer in transfers:
+            source = cls.get_node_or_error(info, transfer["source_id"], only_type="GiftCard")
+            target = cls.get_node_or_error(info, transfer["target_id"], only_type="GiftCard")
+            if source.current_balance_amount < transfer["amount"]:
+                raise ValidationError(
+                    {"amount": ValidationError("Insufficient balance.", code=GiftCardErrorCode.INVALID.value)}
+                )
+            source.current_balance_amount -= transfer["amount"]
+            target.current_balance_amount += transfer["amount"]
+            source.save(update_fields=["current_balance_amount"])
+            target.save(update_fields=["current_balance_amount"])
+            events.gift_card_balance_reset_event(source, None, user, None)
+            events.gift_card_balance_reset_event(target, None, user, None)
+            manager.gift_card_updated(source)
+            manager.gift_card_updated(target)
+            updated.extend([source, target])
+        return GiftCardBalanceTransfer(gift_cards=updated)
